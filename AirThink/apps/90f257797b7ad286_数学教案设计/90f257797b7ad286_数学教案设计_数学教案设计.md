function callAppBridge(opts) {
  return new Promise((resolve, reject) => {
    try {
      if (typeof window.callAppBridge === 'function') {
        const result = window.callAppBridge(opts);
        if (result && typeof result.then === 'function') {
          result.then(resolve).catch(reject);
        } else {
          resolve(result);
        }
      } else {
        // 开发预览兜底（无 bridge 时）
        console.warn('callAppBridge 不可用，使用本地模拟', opts);
        resolve(mockBridge(opts));
      }
    } catch (e) {
      reject(e);
    }
  });
}

// 简易本地模拟，便于纯前端预览
const mockStore = { rows: [] };
function mockBridge(opts) {
  const { type, data, payload } = opts;
  if (type === 'GetRecordCount') return mockStore.rows.length;
  if (type === 'GetTableCSV') {
    const fields = (payload || '').split(',');
    return mockStore.rows.map(r => fields.map(f => (r[f] ?? '').toString().replace(/,/g, '，')).join(',')).join('\n');
  }
  if (type === 'SubmitRecord') {
    const obj = typeof payload === 'string' ? JSON.parse(payload) : payload;
    mockStore.rows.unshift({ ...obj, 教案内容: '' });
    setTimeout(() => {
      mockStore.rows[0].教案内容 = generateMockLesson(obj);
    }, 2500);
    return true;
  }
  if (type === 'GetCellValue') {
    const [field, rowStr] = payload.split('/');
    const row = parseInt(rowStr, 10);
    if (row < 1 || row > mockStore.rows.length) return '';
    return mockStore.rows[row - 1][field] ?? '';
  }
  if (type === 'SetCellValue') {
    const parts = payload.split('/');
    const field = parts[0];
    const row = parseInt(parts[1], 10);
    const val = parts.slice(2).join('/');
    if (row >= 1 && row <= mockStore.rows.length) mockStore.rows[row - 1][field] = val;
    return true;
  }
  if (type === 'DeleteRecord') {
    const row = parseInt(payload, 10);
    if (row >= 1 && row <= mockStore.rows.length) mockStore.rows.splice(row - 1, 1);
    return true;
  }
  if (type === 'GetRecordRow') return 1;
  return null;
}

function escapeHtml(str) {
  if (!str) return '';
  return String(str).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}

function parseCSV(csv) {
  if (!csv || !String(csv).trim()) return [];
  const lines = String(csv).trim().split('\n');
  return lines.map(line => {
    // 简单 CSV 解析（字段中逗号已在写入时替换）
    const cols = [];
    let cur = '', inQ = false;
    for (let i = 0; i < line.length; i++) {
      const c = line[i];
      if (c === '"') { inQ = !inQ; continue; }
      if (c === ',' && !inQ) { cols.push(cur); cur = ''; continue; }
      cur += c;
    }
    cols.push(cur);
    return cols;
  });
}

  }

function startPolling() {
  stopPolling();
  let ticks = 0;
  pollTimer = setInterval(async () => {
    ticks++;
    pollProgress = Math.min(92, 15 + ticks * 6);
    document.getElementById('poll-bar').style.width = pollProgress + '%';
    document.getElementById('poll-msg').textContent = POLL_MSGS[Math.min(ticks, POLL_MSGS.length - 1)];

    try {
      const content = await callAppBridge({
        type: 'GetCellValue',
        data: DATA_SOURCE,
        payload: `教案内容/${currentRow}`
      });
      if (content && String(content).trim().length > 20) {
        stopPolling();
        document.getElementById('poll-bar').style.width = '100%';
        setTimeout(() => {
          document.getElementById('generating-state').classList.add('hidden');
          document.getElementById('result-content').innerHTML = marked.parse(content);
          document.getElementById('result-content').classList.remove('hidden');
          document.getElementById('result-actions').classList.remove('hidden');
          showToast('教案生成完成！');
          loadAllRecords();
        }, 400);
      }
    } catch (e) {
      console.error('poll error', e);
    }

    // 超时保护 ~90s
    if (ticks > 45) {
      stopPolling();
      document.getElementById('generating-state').classList.add('hidden');
      document.getElementById('result-empty').classList.remove('hidden');
      document.getElementById('result-empty').innerHTML = `
        <i class="fas fa-hourglass-half text-4xl mb-3 text-amber-400"></i>
        <p>生成时间较长，请稍后在教案库中查看</p>
        <button class="btn-secondary mt-3" onclick="switchPage('library')">前往教案库</button>`;
    }
  }, 2000);
}

function stopPolling() {
  if (pollTimer) { clearInterval(pollTimer); pollTimer = null; }
}

function copyResult() {
  const rec = allRecords.find(r => r.row === currentRow);
  const text = rec?.教案内容 || document.getElementById('result-content').innerText;
  navigator.clipboard?.writeText(text).then(() => showToast('已复制')).catch(() => showToast('复制失败'));
}

function printResult() {
  const title = document.getElementById('field-课题').value || '数学教案';
  const content = document.getElementById('result-content').innerHTML;
  const w = window.open('', '_blank');
  w.document.write(`<!DOCTYPE html><html><head><title>${escapeHtml(title)}</title>
    <style>body{font-family:sans-serif;max-width:800px;margin:40px auto;line-height:1.7;padding:0 20px}
    h1,h2,h3{color:#1e3a8a}</style></head><body><h1>${escapeHtml(title)}</h1>${content}</body></html>`);
  w.document.close();
  w.print();
}

// ==================== 删除 ====================
function askDelete(row) {
  deleteTargetRow = row;
  document.getElementById('delete-modal').classList.remove('hidden');
}
function closeDeleteModal() {
  deleteTargetRow = null;
  document.getElementById('delete-modal').classList.add('hidden');
}
async function confirmDelete() {
  if (deleteTargetRow == null) return;
  try {
    await callAppBridge({
      type: 'DeleteRecord',
      data: DATA_SOURCE,
      payload: String(deleteTargetRow)
    });
    showToast('已删除');
    closeDeleteModal();
    if (currentRow === deleteTargetRow) {
      currentRow = 0;
      resetForm();
    }
    await loadAllRecords();
    // 删除后行号变化，重新编号
    allRecords.forEach((r, i) => { r.row = i + 1; });
    if (document.getElementById('page-library').classList.contains('active')) filterLibrary();
    else if (document.getElementById('page-home').classList.contains('active')) loadHome();
    else if (document.getElementById('page-detail').classList.contains('active')) switchPage('library');
  } catch (e) {
    console.error(e);
    showToast('删除失败');
  }
}
